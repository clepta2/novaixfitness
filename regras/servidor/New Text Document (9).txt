🛠️ O Guardião da Fortaleza: enforce-security.js
Este script vai ler automaticamente todos os arquivos da sua pasta de regras (scripts/security-rules/) e varrer o seu código-fonte (src/). Se ele encontrar qualquer padrão que quebre uma das suas 130 regras (como um endpoint crítico sem tratamento de IP, uma query sem tratamento atômico no banco ou mídias sem validação por magic numbers), ele trava o fluxo imediatamente.

Crie o arquivo mestre em scripts/enforce-security.js:

JavaScript
const fs = require('fs');
const path = require('path');

// Alinhamento com a Regra 125: Garantir que o ambiente de teste seja limpo
console.log("🛡️ [SECURITY GATE] Inicializando Varredura Dinâmica das 18 Camadas...");

const targetDir = process.argv[2] ? path.resolve(process.argv[2]) : path.resolve(__dirname, '../src');
const rulesDir = path.resolve(__dirname, 'security-rules');

if (!fs.existsSync(targetDir)) {
    console.error(`❌ Erro: Diretório alvo "${targetDir}" não foi encontrado.`);
    process.exit(1);
}

// 1. Carrega todas as regras modulares das 18 camadas em ordem alfabética/numérica
const ruleFiles = fs.readdirSync(rulesDir)
    .filter(file => file.endsWith('.js'))
    .sort();

console.log(`📦 Carregadas ${ruleFiles.length} camadas de validação do terminal.`);

let totalErrors = 0;
let totalFilesScanned = 0;

// Função auxiliar para varrer recursivamente a pasta src/ (React e Node)
function scanDirectory(dir, rules) {
    const files = fs.readdirSync(dir);

    for (const file of files) {
        const fullPath = path.join(dir, file);
        const stat = fs.statSync(fullPath);

        if (stat.isDirectory()) {
            // Pula pastas óbvias de dependências se estiverem no escopo
            if (file !== 'node_modules' && file !== 'dist' && file !== 'build') {
                scanDirectory(fullPath, rules);
            }
        } else if (stat.isFile() && /\.(js|jsx|ts|tsx)$/.test(file)) {
            totalFilesScanned++;
            const content = fs.readFileSync(fullPath, 'utf8');

            // Executa cada camada de regra em cima do arquivo de código atual
            rules.forEach(ruleModule => {
                const results = ruleModule.validate(fullPath, content);
                if (results && results.errors.length > 0) {
                    results.errors.forEach(err => {
                        totalErrors++;
                        console.error(`⚠️ [CAMADA ${ruleModule.layer}] Falha na Regra [${err.ruleId}] em: ${fullPath}:${err.line}`);
                        console.error(`   👉 Motivo: ${err.message}\n`);
                    });
                }
            });
        }
    }
}

// 2. Inicializa e mapeia os módulos das regras
const activeRules = ruleFiles.map(file => {
    const rulePath = path.join(rulesDir, file);
    const loaded = require(rulePath);
    return {
        filename: file,
        layer: file.split('-')[2] || 'Geral', // Pega o nome do módulo
        validate: loaded.validate
    };
});

// 3. Executa a varredura cirúrgica
console.log(`🔍 Vasculhando arquivos em: ${targetDir}...`);
scanDirectory(targetDir, activeRules);

console.log(`\n==================================================`);
console.log(`📊 RELATÓRIO FINAL DO SECURITY GATE:`);
console.log(`   📂 Arquivos de código analisados: ${totalFilesScanned}`);
console.log(`   🚨 Infrações críticas detectadas: ${totalErrors}`);
console.log(`==================================================\n`);

// Se houver qualquer falha, quebra o processo e impede o Git Commit / Docker Build
if (totalErrors > 0) {
    console.error("❌ [BLOQUEADO] O código viola as diretrizes de segurança estabelecidas.");
    process.exit(1);
} else {
    console.log("✅ [APROVADO] Todas as 18 camadas estão íntegras. Código blindado!");
    process.exit(0);
}
🧱 Exemplo de Módulo de Regra: 106-110-checkout-security.js
Para que o validador mestre funcione, cada arquivo dentro de scripts/security-rules/ deve exportar uma função validate.

Aqui está o exemplo da sua Camada 14 (Checkout), aplicando a regra de que dados brutos de cartão nunca podem tocar o backend (Regra 106) e a trava contra cliques duplos no checkout usando o Redis (Regra 108).

Crie em scripts/security-rules/106-110-checkout-security.js:

JavaScript
module.exports = {
    validate: function (filePath, content) {
        const errors = [];
        const lines = content.split('\n');

        // Padrões perigosos que violam as regras da Camada 14
        const rawCardRegex = /(cardNumber|card_number|cvv|cardCvc|expirationDate)/i;
        const checkoutRouteRegex = /\/checkout|\/pagamento|\/charge/i;

        lines.forEach((lineContent, index) => {
            const lineNumber = index + 1;

            // Validação da Regra 106 (Proibir captura de cartão cru no Backend)
            if (filePath.includes('/backend/') && rawCardRegex.test(lineContent)) {
                // Ignora se for apenas um comentário legítimo de aviso
                if (!lineContent.includes('//') && !lineContent.includes('*')) {
                    errors.push({
                        line: lineNumber,
                        ruleId: "R106",
                        message: "Dados brutos de cartão de crédito detectados no backend. Use Tokenização Direta via SDK do Asaas no front-end!"
                    });
                }
            }

            // Validação da Regra 108 (Trava de concorrência com Redis no Checkout)
            if (checkoutRouteRegex.test(lineContent) && filePath.includes('/routes/')) {
                if (!content.includes('redis') && !content.includes('lock')) {
                    errors.push({
                        line: lineNumber,
                        ruleId: "R108",
                        message: "Rota de pagamento detectada sem mecanismo de trava/timeout ativo no Redis. Risco de cobrança duplicada!"
                    });
                }
            }
        });

        return { errors };
    }
};
